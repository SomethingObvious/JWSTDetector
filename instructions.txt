source .venv/bin/activate

find . -name "*:Zone.Identifier" -type f -delete


# ---------------------------------------------------------------------------
# 1. Tiles
# ---------------------------------------------------------------------------

# Big data: smaller tiles, tighter coverage cut.
python prep_data.py \
  --sensor miri \
  --indir rawfits \
  --out datasets/miri \
  --pattern "*i2d.fits" \
  --sci_ext SCI \
  --wht_ext WHT \
  --tile_size 448 --upscale 1 --stride 420 \
  --min_wht_frac 0.80 \
  --source_sigma 5 --min_source_frac 2e-4 \
  --p_lo 1 --p_hi 99.8 --asinh_q 10.0 \
  --stretch_scope mosaic \
  --wcs_in_name

# Small data.
python prep_data.py \
  --sensor miri \
  --indir rawfits \
  --out datasets/miri \
  --pattern "*sci.fits" \
  --sci_ext SCI \
  --wht_ext WHT \
  --tile_size 672 --upscale 1 --stride 644 \
  --min_wht_frac 0.5 \
  --source_sigma 5 --min_source_frac 2e-4 \
  --p_lo 1 --p_hi 99.8 --asinh_q 10.0 \
  --stretch_scope mosaic \
  --wcs_in_name

# Check what the thresholds would keep before writing anything.
python prep_data.py --sensor miri --indir rawfits --out datasets/miri \
  --pattern "*sci.fits" --tile_size 672 --stride 644 --min_wht_frac 0.5 --dry-run --limit 2


# ---------------------------------------------------------------------------
# 2. Scoring
# ---------------------------------------------------------------------------

# Small data: ViT-L at 672, eight-fold augmented reference set.
python run_query_bootstrap.py \
  --data_root datasets/miri \
  --query_subdir query \
  --recursive_query \
  --model_name dinov3-vitl16-pretrain-sat493m \
  --resolution 672 \
  --batch_size 16 \
  --num_workers 8 \
  --knn_metric L2_normalized \
  --k_neighbors 20 \
  --max_bank_patches 1000000 \
  --score_mode top1p \
  --rotation \
  --save_examples \
  --save_patch_dists \
  --save_tiffs \
  --device cuda:0 \
  --seed 0 \
  --init_ref_frac 0.2 \
  --init_ref_min 100 \
  --init_ref_max 425 \
  --bootstrap_keep_frac 0.1 \
  --bootstrap_keep_min 100 \
  --bootstrap_keep_max 425 \
  --out_dir results_query_only \
  --tag miri_672_big
